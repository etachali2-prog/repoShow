async function fetchSheetData() { try { const response = await fetch(SCRIPT_URL); const data = await response.json(); if (data.length === 0) { document.getElementById('statusMsg').innerText = "No records found in the sheet."; return; } // Hide loading status, show table document.getElementById('statusMsg').style.display = 'none'; document.getElementById('dataTable').style.display = 'table'; // Render Table Headers const tableHeader = document.getElementById('tableHeader'); const headers = Object.keys(data[0]); headers.forEach(header => { const th = document.createElement('th'); th.innerText = header; tableHeader.appendChild(th); }); // Helper function to extract only YYYY-MM-DD function formatDateOnly(value) { if (typeof value === 'string' && value.match(/^\d{4}-\d{2}-\d{2}T/)) { return value.split('T')[0]; } return value !== undefined ? value : ''; } // Render Table Body const tableBody = document.getElementById('tableBody'); data.forEach(row => { const tr = document.createElement('tr'); headers.forEach(header => { const td = document.createElement('td'); td.innerText = formatDateOnly(row[header]); tr.appendChild(td); }); tableBody.appendChild(tr); }); } catch (error) { console.error("Error fetching data:", error); document.getElementById('statusMsg').innerText = "Failed to load sheet data. Please check your Web App URL."; } }
